cmake_minimum_required(VERSION 3.24)
project(miracl-wrapper VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 1. Dependency Handling
# ---------------------------------------------------------
# Remove cmake module path as we are compiling from source
# list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# IMPORTANT: Enter the external directory; ensure external/CMakeLists.txt includes GMP
add_subdirectory(external)

# [DELETED] find_package(GMP REQUIRED)
# We no longer search for system libraries; using targets built via add_subdirectory instead

# 2. Library Definition
# ---------------------------------------------------------
add_library(WrapperLib STATIC
        src/Tools.cpp
)

# 3. Header Include Paths
# ---------------------------------------------------------
target_include_directories(WrapperLib
        PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
        PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# 4. Link Dependencies
# ---------------------------------------------------------
target_link_libraries(WrapperLib PUBLIC
        miracl_core
        # Link the Target name generated by the submodule.
        # NOTE: Verify the actual Target name produced by the rookie-papers/GMP submodule.
        # If it follows standard CMake conventions, it is typically 'gmp' or 'gmp::gmp'.
        # Assuming the generated static library target is named 'gmp'
        gmp
        # If the submodule does not generate gmpxx, this line may need to be removed or
        # check if the submodule supports C++ bindings.
        # gmpxx
)

enable_testing()
